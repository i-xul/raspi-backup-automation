# ==========================================================
# Ubuntu Desktop Restic Exclude List
#
# This file contains paths that should never be included in
# system backups.
#
# The goal is to avoid backing up virtual filesystems,
# temporary runtime data, caches, and the backup repository
# itself.
# ==========================================================


# ----------------------------------------------------------
# Virtual filesystems
# ----------------------------------------------------------

/proc
/sys
/dev
/run


# ----------------------------------------------------------
# Temporary files
# ----------------------------------------------------------

/tmp
/var/tmp


# ----------------------------------------------------------
# Runtime locks and sockets
# ----------------------------------------------------------

/var/run


# ----------------------------------------------------------
# Lost+found (filesystem metadata)
# ----------------------------------------------------------

/lost+found


# ----------------------------------------------------------
# Backup destination
#
# Prevent recursive backups of the repository itself.
# ----------------------------------------------------------

/mnt/nas-backups


# ----------------------------------------------------------
# User trash
# ----------------------------------------------------------

/home/*/.local/share/Trash

# ----------------------------------------------------------

# ----------------------------------------------------------

*.part
